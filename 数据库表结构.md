## 用户列表模块

### 用户表

```sql
CREATE TABLE IF NOT EXISTS user  (
    `id` bigint NOT NULL PRIMARY KEY COMMENT '用户ID，全局唯一ID，不可修改',
    `username` varchar(30) NOT NULL COMMENT '用户名，30，可重复',
    `password` varchar(255) NOT NULL DEFAULT '' COMMENT '密码，密文',
    introduction VARCHAR(30) default '' COMMENT '用户简介,30',
    `email` varchar(50) DEFAULT '' COMMENT '用户邮箱',
    `mobile` varchar(11) DEFAULT '' COMMENT '手机号码',
    `gender` tinyint DEFAULT 0 COMMENT '用户性别，0-女，1-男，2-未知',
    `avatar` varchar(512) DEFAULT '' COMMENT '头像地址',
    `status` tinyint NOT NULL DEFAULT 0 COMMENT '帐号状态（0正常 1异常 2封禁 3注销）',
    ban_time datetime comment '封禁时间，表示到这时进行解封',

    created_time datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '注册时间',
    updated_by bigint DEFAULT 0 COMMENT '更新者',
    updated_time datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    deleted tinyint NOT NULL DEFAULT 0 COMMENT '是否删除，0-否，1-是'
    
) ENGINE = InnoDB CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci COMMENT = '用户信息表';
```





## 会话相关

### 会话表

```sql
CREATE TABLE IF NOT EXISTS conversation (
    id bigint NOT NULL PRIMARY KEY comment '会话的唯一标识符',
    user_id BIGINT NOT NULL COMMENT '用户ID，关联用户表',
    name varchar(80) comment '该会话的名称',
    context_summary varchar(200) comment '上下文概括',
    current_round int not null default 1 comment '当前轮次',
    
    created_time datetime DEFAULT CURRENT_TIMESTAMP COMMENT '记录创建时间',
    created_by bigint COMMENT '创建人用户ID',
    updated_time datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '最后更新时间',
    deleted tinyint NOT NULL DEFAULT 0 COMMENT '是否删除，0-否，1-是',
    
    index idx_user_id_updated_time (user_id, updated_time) comment '用于查询排序'
    
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='会话表';
```

### 情感诊断书表

用于本次对应的全面总结，根据用户会话的全部内容进行用户本人情感的全面分析。主要强调用户当前的情感诊断（类似心理治疗后的诊断书）。属于会话级的情感分析。一个会话只有一个。

```sql
CREATE TABLE IF NOT EXISTS emotion_diagnosis (
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '诊断书主键ID',
    conversation_id BIGINT UNIQUE NOT NULL COMMENT '会话ID',
    user_id BIGINT NOT NULL COMMENT '用户ID（冗余存储，便于单独查询）',
    round_num int not null default 1 comment '该诊断数据创建或更新时的轮次',
    
    -- 1. 核心情绪结论维度（原基础上扩展）
    diagnosis_content VARCHAR(1000) DEFAULT '' COMMENT '诊断书核心内容（自然语言总结）',
    core_emotion_label VARCHAR(32) DEFAULT '' COMMENT '核心情绪标签（如焦虑/抑郁/开心/中性）',
    core_emotion_conf_avg DECIMAL(4,3) DEFAULT 0.000 COMMENT '核心情绪平均置信度（0-1，精度提升）',
    core_emotion_intensity VARCHAR(16) DEFAULT '' COMMENT '核心情绪强度（轻度/中度/重度/极重度）',
    secondary_emotion_labels VARCHAR(128) DEFAULT '' COMMENT '次要情绪标签（多个用逗号分隔，如"烦躁,委屈,孤独"）',
    secondary_emotion_conf VARCHAR(128) DEFAULT '' COMMENT '次要情绪置信度（与次要标签一一对应，如"0.85,0.72,0.68"）',
    
    -- 2. 情绪分布维度（支撑饼图/占比类图表）
    negative_emotion_ratio DECIMAL(3,2) DEFAULT 0.00 COMMENT '负向情绪占比（0-1）',
    positive_emotion_ratio DECIMAL(3,2) DEFAULT 0.00 COMMENT '正向情绪占比（0-1）',
    neutral_emotion_ratio DECIMAL(3,2) DEFAULT 0.00 COMMENT '中性情绪占比（0-1，三者和为1）',
    negative_emotion_detail VARCHAR(255) DEFAULT '' COMMENT '负向情绪细分占比（如"焦虑:0.45,愤怒:0.25,悲伤:0.10"）',
    positive_emotion_detail VARCHAR(255) DEFAULT '' COMMENT '正向情绪细分占比（如"开心:0.30,欣慰:0.15,放松:0.05"）',
    
    -- 3. 情绪动态变化维度（支撑折线图/趋势类图表）
    emotion_trend VARCHAR(30) DEFAULT '' COMMENT '整体情绪趋势',
    emotion_peak_round INT DEFAULT 0 COMMENT '情绪峰值轮次（核心情绪强度最高的轮次）',
    emotion_valley_round INT DEFAULT 0 COMMENT '情绪低谷轮次（核心情绪强度最低的轮次）',
    emotion_fluctuation_amplitude DECIMAL(4,3) DEFAULT 0.000 COMMENT '情绪波动幅度（峰值-谷值的置信度差）',
    emotion_stable_rounds INT DEFAULT 0 COMMENT '情绪平稳的轮次数量（占总轮次的辅助指标）',
    
    -- 4. 情绪触发因素维度（支撑归因分析）
    core_trigger_scene VARCHAR(64) DEFAULT '' COMMENT '核心触发场景（如工作压力/人际关系/家庭矛盾）',
    core_trigger_keywords VARCHAR(128) DEFAULT '' COMMENT '核心触发关键词（多个用逗号分隔，如"加班,吵架,失业"）',
    trigger_round_num INT DEFAULT 0 COMMENT '首次出现核心触发因素的轮次',
    
    -- 5. 风险与干预维度（支撑决策/建议类展示）
    emotion_risk_level VARCHAR(16) DEFAULT '' COMMENT '情绪风险等级（低/中/高/危急）',
    emotion_adjust_suggestion VARCHAR(500) DEFAULT '' COMMENT '情绪调节建议（自然语言，如"建议适当休息，减少加班频率"）',
    need_manual_intervene TINYINT DEFAULT 0 COMMENT '是否需要人工干预（0-否，1-是）',
    
    -- 6. 基础元数据
    created_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '记录创建时间',
    updated_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '最后更新时间',
    created_by BIGINT DEFAULT 0 COMMENT '创建人用户ID',
    deleted TINYINT NOT NULL DEFAULT 0 COMMENT '是否删除，0-否，1-是',
    
    -- 索引优化（适配多维度查询）
    INDEX idx_conversation_id (conversation_id)
    
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='情感诊断书表（多轮会话汇总）';
```





### 情绪分析表

用于在每次对话后进行生成，主要用于结合上下文信息分析分析本次对话时，用户的情感状态，便于用户查询自身情感的变化。属于对话级的情感分析。一次对话生成一个，不进行修改。

```sql
CREATE TABLE IF NOT EXISTS emotion_analysis (
    `id` bigint NOT NULL PRIMARY KEY AUTO_INCREMENT COMMENT '自增id',
    conversation_id bigint NOT NULL COMMENT '会话ID',
    round_num int default 1 comment '对应的轮次，与会话ID结合查询对应的对话信息',
    
    analysis_content varchar(100) default '' comment '情感分析详情',
    `emotion_label` varchar(32) default '' COMMENT '情感标签（如anger/开心/neutral/不满等）',
    `emotion_sub_label` varchar(32) DEFAULT '' COMMENT '情感细分标签（如愤怒可细分“不满/暴怒/抱怨”）',
    `emotion_score` decimal(3,2) DEFAULT NULL COMMENT '情感识别置信度（0-1，如0.9200）',
    emotion_trend VARCHAR(30) COMMENT '情绪变化趋势',
    negative_emotion_ratio DECIMAL(3,2) COMMENT '负向情绪占比（0-1）',
    positive_emotion_ratio decimal(3,2) DEFAULT 0.00 COMMENT '正向情绪占比（0-1）',
    
    created_time datetime DEFAULT CURRENT_TIMESTAMP COMMENT '记录创建时间',
    created_by bigint COMMENT '创建人用户ID',
    deleted tinyint NOT NULL DEFAULT 0 COMMENT '是否删除，0-否，1-是',
    
    index idx_conversation_id_created_time (conversation_id, created_time) comment '用于查询排序'
    
) ENGINE = InnoDB AUTO_INCREMENT = 286 CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci COMMENT = '情绪记录表';
```
